[32m+[m[32m/**[m
[32m+[m[32m * @brief Fetches the first event pending in the event pool.[m

[32m+[m[32m * enclosing critical section when needed.[m
[32m+[m[32m * @param [in] p_obj The pointer to the self object.[m
[32m+[m[32m * @returns true, when an event was fetched from the event pool and is waiting[m
[32m+[m[32m * for processing.[m
[32m+[m[32m * @returns false, when the event pool is empty, or the p_obj is NULL.[m
[32m+[m[32m */[m
[32m+[m[32mbool[m
[32m+[m[32msafesm_sm1_fetch_event(safesm_sm1_t* const p_obj)[m
[32m+[m[32m{[m
[32m+[m[32m    bool b_is_new_event = false;[m
[32m+[m
[32m+[m[32m    if(NULL != p_obj)[m
[32m+[m[32m    {[m

[32m+[m[32m                    (event_pool_locator_t)[m

[32m+[m[32m/**[m
[32m+[m[32m * @brief Executes the proper action corresponding to the event fetched from the[m
[32m+[m[32m * event pool.[m
[32m+[m[32m * @details This function was separated from the other two (*fetch* and[m
[32m+[m[32m * *release*) functions, because in normal scenario this function doesn't need[m
[32m+[m[32m * to be called from a critical section. Since the event at the head of the[m
[32m+[m[32m * event pool has already been fetched, no action on the event pool is able to[m

[32m+[m[32m * pool, there is no need for an additional temporary buffer for keeping a copy[m
[32m+[m[32m * of the event data (which could be troublesome due to separate data types for[m
[32m+[m[32m * each event).[m

[32m+[m[32m * 'event_pool.fetched_event' field. This approach saves memory and has no need[m

[32m+[m[32m * event data due to retrieving information from the 'event_pool.fetched_event'[m
[32m+[m[32m * field.[m
[32m+[m[32m * @returns true when a fetched event was waiting for processing.[m
[32m+[m[32m * @returns false when there was no event waiting for processing or the p_obj[m
[32m+[m[32m * was NULL.[m
[32m+[m[32m */[m
[32m+[m[32mbool[m

[32m+[m[32m{[m
[32m+[m[32m    bool b_is_new_event = false;[m
[32m+[m
[32m+[m[32m    if ((NULL != p_obj)[m
[32m+[m[32m        && (SM_EVENT_STATUS_PENDING == p_obj->event_pool.event_proc_status))[m
[32m+[m[32m    {[m
[32m+[m[32m        sm_event_status_t temp_status = SM_EVENT_STATUS_IGNORED;[m
[32m+[m
[32m+[m[32m        switch(p_obj->event_pool.fetched_event.event_type)[m
[32m+[m[32m        {[m
[32m+[m[32m            case SAFESM_SM1_A:[m
[32m+[m[32m                temp_status = safesm_sm1_dispatch_a(p_obj);[m
[32m+[m[32m                break;[m
[32m+[m[32m            default:[m
[32m+[m[32m                break;[m
[32m+[m[32m        }[m
[32m+[m
[32m+[m[32m        p_obj->event_pool.event_proc_status = temp_status;[m
[32m+[m[32m        b_is_new_event = true;[m
[32m+[m[32m    }[m
[32m+[m
[32m+[m[32m    return b_is_new_event;[m
[32m+[m[32m}[m

[32m+[m[32m/**[m
[32m+[m[32m * @brief Removes the processed event from the event pool.[m
[32m+[m[32m * @details This function is separated to allow the user to call it from an[m
[32m+[m[32m * enclosing critical section when needed.[m
[32m+[m[32m * @param [in] p_obj The pointer to the self object of the sm1 class.[m
[32m+[m[32m * @returns true, if the last fetched event had already been processed and then[m
[32m+[m[32m * removed from the event pool.[m
[32m+[m[32m * @returns false if the p_obj is NULL or there was no processed event in the[m
[32m+[m[32m * event pool.[m
[32m+[m[32m */[m
[32m+[m[32mbool[m
[32m+[m[32msafesm_sm1_release_event(safesm_sm1_t* const p_obj)[m
[32m+[m[32m{[m
[32m+[m[32m    bool b_is_released = false;[m
[32m+[m
[32m+[m[32m    if ((NULL != p_obj)[m
[32m+[m[32m        && (SM_EVENT_STATUS_PENDING < p_obj->event_pool.event_proc_status)[m
[32m+[m[32m        && (SM_EVENT_STATUS_DISPATCHED > p_obj->event_pool.event_proc_status))[m
[32m+[m[32m    {[m
[32m+[m[32m        b_is_released = event_pool_dequeue(&p_obj->event_pool.manager);[m

[32m+[m[32m        if(b_is_released)[m
[32m+[m[32m        {[m
[32m+[m[32m            p_obj->event_pool.event_proc_status = SM_EVENT_STATUS_DISPATCHED;[m
[32m+[m[32m        }[m
[32m+[m[32m    }[m
[32m+[m
[32m+[m[32m    return b_is_released;[m
[32m+[m[32m}[m
